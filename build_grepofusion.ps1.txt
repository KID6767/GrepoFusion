function Log([string]$msg){ 
  $ts = (Get-Date).ToString('yyyy-MM-dd HH:mm:ss')
  Write-Host "$ts  $msg" -ForegroundColor Cyan
}


function EnsureDir($path){ 
  if(-not (Test-Path $path)){ 
    New-Item -ItemType Directory -Path $path | Out-Null 
  }
}


function WriteUtf8([string]$path,[string]$content){
  $dir = Split-Path -Parent $path; EnsureDir $dir
  [IO.File]::WriteAllText($path, $content, (New-Object System.Text.UTF8Encoding($false)))
}


function WriteBytes([string]$path,[byte[]]$bytes){
  $dir = Split-Path -Parent $path; EnsureDir $dir
  [IO.File]::WriteAllBytes($path,$bytes)
}


# bezpieczny dekoder Base64
function SafeB64([string]$b64){
  try { return [Convert]::FromBase64String($b64) } 
  catch { return @() }
}

    queue:[],
    load:function(){
      try{
        this.queue = JSON.parse(localStorage.getItem(this.key)||"[]");
      }catch(e){ this.queue=[]; }
    },
    save:function(){
      localStorage.setItem(this.key,JSON.stringify(this.queue));
    },
    add:function(building,targetLevel){
      this.queue.push({building,targetLevel});
      this.save();
      alert("Dodano do kolejki: "+building+" → "+targetLevel);
    },
    run:function(){
      this.load();
      console.log("GF_Helper_Build run:",this.queue);
      const res = {wood:unsafeWindow.Game.resources.wood,
                   stone:unsafeWindow.Game.resources.stone,
                   iron:unsafeWindow.Game.resources.iron,
                   pop:unsafeWindow.Game.current_population};
      for(const job of this.queue){
        const b = unsafeWindow.Buildings.getBuildingData(job.building);
        if(!b) continue;
        if(b.level>=job.targetLevel) continue;
        const cost = unsafeWindow.Buildings.buildingCosts(job.building,b.level+1);
        if(res.wood>=cost.wood && res.stone>=cost.stone && res.iron>=cost.iron && res.pop>=cost.population){
          console.log("Buduję:",job.building,"lvl",b.level+1);
          unsafeWindow.Buildings.build(job.building);
          res.wood-=cost.wood; res.stone-=cost.stone; res.iron-=cost.iron; res.pop-=cost.population;
        }
      }
    }

$HelperBuildLogic = @'
// ===== Pomocnik budowlańca (Senat) =====
(function(){
  "use strict";
  const log=(...a)=>console.log("%c[GF::Build]","color:#d4af37",...a);


  // profil rozbudowy (z grubsza bezpieczny dla fresh gry)
  const PLAN = [
    { main:10, farm:10, storage:10, lumber:10, stoner:10, ironer:10 },
    { main:20, farm:20, storage:20, lumber:20, stoner:20, ironer:20, barracks:5, academy:13 },
    { temple:15, market:10, hide:10, wall:5, docks:5 },
    { main:25, farm:35, storage:30, lumber:30, stoner:30, ironer:30, barracks:20, docks:15 },
    { academy:36, wall:20, market:20, temple:30 }
  ];


  function ready(){
    return !!(window.ITowns && window.MM && window.gpAjax && window.Game);
  }
  function has(res, need){
    return res.wood>=need.wood && res.stone>=need.stone && res.iron>=need.iron;
  }
  function enqueue(townId, building){
    return new Promise((resolve,reject)=>{
      gpAjax.ajaxPost("frontend_bridge","execute",{
        model_url:"BuildingOrder",
        action_name:"buildUp",
        arguments:{building_id: building},
        town_id: townId
      }, false, { success:resolve, error:reject });
    });
  }
  async function tryTown(model){
    const a = model.attributes || model.get && model.get("attributes");
    const id = a.id;
    if(a.is_building_order_queue_full) return;
    const bd = a.building_data;
    const res = ITowns.towns[id].resources();
    // znajdź pierwszy pasujący cel
    for(const layer of PLAN){
      for(const [name, maxLv] of Object.entries(layer)){
        const d = bd[name];
        if(!d || d.has_max_level) continue;
        const next = d.next_level;
        if(next>maxLv) continue;
        if(!has(res, d.resources_for)) continue;
        try{
          await enqueue(id, name);
          log("enqueue", name, "→", next, "@", ITowns.towns[id].name);
          return; // jeden wpis na pętlę
        }catch(e){ /* ignoruj */ }
      }
    }
  }
  async function loop(){
    if(!ready()) return setTimeout(loop, 3000);
    try{
      const all = Object.values(MM.getModels().BuildingBuildData||{});
      for(const m of all) await tryTown(m);
    }catch(e){}
    setTimeout(loop, 45000 + Math.random()*15000);
  }
  window.GF_Helper_Build = { run: loop };
  setTimeout(loop, 2500);
})();
'@

$HelperRecruitLogic = @'
// ===== Rekruter (Koszary/Port) =====
(function(){
  "use strict";
  const log=(...a)=>console.log("%c[GF::Recruit]","color:#d4af37",...a);


  // Plan minimalny (na start bezpieczny): łódź transportowa i proce/łucznicy
  const PLAN = {
    barracks: [ ["slinger", 20], ["archer", 10] ],
    harbor:   [ ["transport_ship", 5], ["bireme", 5] ]
  };


  function ready(){
    return !!(window.ITowns && window.MM && window.gpAjax && window.Game);
  }


  function enqueue(townId, kind, unit, amount){
    const model = kind==="barracks" ? "BarracksOrder" : "DocksOrder";
    const act   = "build";
    const args  = kind==="barracks" ? {unit_id:unit, amount:amount} : {unit_id:unit, amount:amount};
    return new Promise((resolve,reject)=>{
      gpAjax.ajaxPost("frontend_bridge","execute",{
        model_url: model,
        action_name: act,
        arguments: args,
        town_id: townId
      }, false, { success:resolve, error:reject });
    });
  }


  async function doTown(townId){
    // barracks
    for(const [unit, amt] of PLAN.barracks){
      try{ await enqueue(townId,"barracks",unit,amt); log("barracks",unit,amt,"@",ITowns.towns[townId].name) }catch(e){}
      await new Promise(r=>setTimeout(r,700));
    }
    // harbor
    for(const [unit, amt] of PLAN.harbor){
      try{ await enqueue(townId,"harbor",unit,amt); log("harbor",unit,amt,"@",ITowns.towns[townId].name) }catch(e){}
      await new Promise(r=>setTimeout(r,700));
    }
  }


  async function loop(){
    if(!ready()) return setTimeout(loop, 3000);
    try{
      const ids = Object.keys(ITowns.towns||{});
      for(const id of ids) await doTown(+id);
    }catch(e){}
    setTimeout(loop, 90000 + Math.random()*30000);

$AssetOverrideJs = @"
(function(){
  'use strict';
  const RAW_MAP = __ASSET_MAP__ ; // JSON → obiekt


  function mapSrc(src){
    try{
      // statki
      for(const [k,v] of Object.entries(RAW_MAP.units||{})){
        if(src.includes(k)) return (v.pirate||v.classic||src);
      }
      // budynki
      for(const [k,v] of Object.entries(RAW_MAP.buildings||{})){
        if(src.includes(k)) return v;
      }
    }catch(e){}
    return src;
  }


  // patch createElement → IMG.setAttribute('src',...)
  const _create=document.createElement;
  document.createElement=function(tag){
    const el=_create.call(document,tag);
    if((''+tag).toLowerCase()==='img'){
      const _set=el.setAttribute;
      el.setAttribute=function(n,val){
        if(n==='src' && typeof val==='string'){ val=mapSrc(val) }
        return _set.call(this,n,val);
      }
    }
    return el;
  };


  // patch istniejących obrazów
  const patch=(root)=>{
    root.querySelectorAll && root.querySelectorAll('img[src]').forEach(img=>{
      img.src = mapSrc(img.src);
    });
  };


  const mo=new MutationObserver(muts=>{
    muts.forEach(m=>{
      m.addedNodes && m.addedNodes.forEach(n=>{
        if(n.nodeType===1) patch(n);
      });
    });
  });
  mo.observe(document.documentElement,{childList:true,subtree:true});
  patch(document);
})();

$AssetMapManagerJs = @'
(function(){
  "use strict";
  if(window.GF_AssetMap) return;


  const listeners = [];
  const emit = ()=> listeners.forEach(fn=>{ try{ fn(window.GF_ASSET_MAP||{}) }catch(_){}});


  window.GF_AssetMap = {
    /** ustawia nową mapę (obiekt JSON) i triggeruje przebudowę obrazów **/
    set(map){
      try{
        window.GF_ASSET_MAP = map || {};
        console.log("[GF::AssetMap] applied:", Object.keys(window.GF_ASSET_MAP||{}).length, "groups");
        emit();
        // Przepatchuj wszystkie obrazki
        document.querySelectorAll("img[src]").forEach(img=>{
          try{
            const s = window.GF_AssetMap.mapSrc(img.src);
            if(s && s!==img.src) img.src = s;
          }catch(_){}
        });
      }catch(e){ console.warn("[GF::AssetMap] set error", e); }
    },
    /** wczytuje mapę z URL (np. raw z GitHuba) **/
    async loadFromUrl(url){
      try{
        const res = await fetch(url, {cache:"no-store"});
        const json = await res.json();
        this.set(json);
        return json;
      }catch(e){
        console.warn("[GF::AssetMap] loadFromUrl error", e);
        throw e;
      }
    },
    /** łączy obecną mapę z nową (shallow) **/
    merge(partial){
      const base = window.GF_ASSET_MAP || {};
      const out = Object.assign({}, base);
      for(const k in partial){ out[k] = Object.assign({}, base[k]||{}, partial[k]); }
      this.set(out);
    },
    /** liczy podmiany (prosty raport) **/
    count(){
      const m = window.GF_ASSET_MAP||{};
      const u = Object.keys(m.units||{}).length;
      const b = Object.keys(m.buildings||{}).length;
      const ui= Object.keys(m.ui||{}).length;
      return {units:u, buildings:b, ui};
    },
    /** mapowanie pojedynczego src (używane przez override) **/
    mapSrc(src){
      try{
        const M = window.GF_ASSET_MAP||{};
        // units
        const units = M.units||{};
        for(const k in units){
          if(src.includes(k)){
            const v = units[k];
            return v.pirate || v.remaster || v.classic || v.dark || v; // preferencja stylów
          }
        }
        // buildings
        const b = M.buildings||{};
        for(const k in b){ if(src.includes(k)) return b[k]; }
        // ui
        const ui = M.ui||{};
        for(const k in ui){ if(src.includes(k)) return ui[k]; }
      }catch(_){}
      return src;
    },
    onChange(fn){ listeners.push(fn); }
  };
})();

# —— 2) Weryfikacja integralności userscripta
function Verify-Build {
  param([string]$Path)


  if(-not (Test-Path $Path)){
    throw "Brak pliku userscripta: $Path"
  }


  $txt = Get-Content -Raw -Path $Path


  # Minimalne sygnatury, które MUSZĄ być obecne
  $needles = @(
    'GF_AssetMap',           # menedżer mapy assetów (live)
    'GF_ASSET_MAP',          # runtime mapa
    'gf-theme-style',        # wstrzyknięty style tag motywu
    'aegis-config',          # panel konfiguracji
    'GF_Helper_Build',       # helper: senat
    'GF_Helper_Recruit',     # helper: koszary/port
    'GF_Helper_Academy',     # helper: akademia
    'gf-changelog'           # changelog overlay
  )


  foreach($n in $needles){
    if($txt -notmatch [Regex]::Escape($n)){
      throw "Weryfikacja nie powiodła się – brak znacznika: $n"
    }
  }


  # Prosta kontrola błędnych here-stringów (niedomknięte @"… "@)
  $open  = ($txt | Select-String -Pattern '@"' -AllMatches).Matches.Count
  $close = ($txt | Select-String -Pattern '"@' -AllMatches).Matches.Count
  if($open -ne $close){
    throw "Here-stringi niesymetryczne: @`" ($open) vs `"`@ ($close)"
  }


  # SHA256
  $sha = (Get-FileHash -Algorithm SHA256 $Path).Hash
  return $sha
}

# —— 7) Paczka ZIP (Part10) – nadpisz poprzednią, jeśli była
try{
  $ZipOut = Join-Path $RepoRoot ("GrepoFusion-$VersionGF.zip")
  if(Test-Path $ZipOut){ Remove-Item $ZipOut -Force -ErrorAction SilentlyContinue }


  $pkgDir = Join-Path $RepoRoot "_pkg_$([Guid]::NewGuid().ToString('N'))"
  EnsureDir $pkgDir


  Copy-Item $Assets      -Destination $pkgDir -Recurse
  Copy-Item $Userscripts -Destination $pkgDir -Recurse
  Copy-Item $Docs        -Destination $pkgDir -Recurse
  Copy-Item $Dist        -Destination $pkgDir -Recurse
  Copy-Item $Tools       -Destination $pkgDir -Recurse


  Add-Type -AssemblyName 'System.IO.Compression.FileSystem'
  [IO.Compression.ZipFile]::CreateFromDirectory($pkgDir, $ZipOut)
  Remove-Item $pkgDir -Recurse -Force


  Log "ZIP (Part10): $ZipOut"
  Log ("SHA-256: " + (Get-FileHash -Algorithm SHA256 $ZipOut).Hash)
}catch{
  Write-Host ("ZIP ERR: "+ $_.Exception.Message) -ForegroundColor Red
  throw
}


# —— 8) Git commit + tag (vX.Y.Z-rcN jeśli w wersji draft) i push
try{
  & git add -A | Out-Null


  $tagName = "v$VersionGF"
  $msg = "GrepoFusion $VersionGF: verified build, report, installers, live assetmap, helpers status"


  & git commit -m $msg | Out-Null
  # tag nadpisywany (annotated); jeśli istnieje – przestawiamy
  if((& git tag --list $tagName)){
    & git tag -d $tagName | Out-Null
  }
  & git tag -a $tagName -m $msg | Out-Null
  & git push | Out-Null
  & git push origin $tagName | Out-Null


  Log "git push + tag $tagName ✓"
}catch{
  Write-Host ("GIT WARN: "+$_.Exception.Message) -ForegroundColor Yellow
}
